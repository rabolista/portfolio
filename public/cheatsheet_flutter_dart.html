Variables & Null Safety dart basics
var, final, const & Null Safety Interview Fav
Dart is sound null-safe by default since Dart 2.12
void main() {
  // var — type inferred, can be reassigned
  var name = 'Robert';

  // final — set once at runtime
  final age = 35;

  // const — compile-time constant
  const pi = 3.14159;

  // explicit types
  String role = 'BI Developer';
  int years = 15;

  // null safety — a type is non-nullable unless marked with ?
  String? nickname; // nullable — defaults to null
  String greeting = nickname ?? 'friend'; // ?? = null-coalescing

  print('$name, $age, $role, $years, $greeting');
}
Output
Robert, 35, BI Developer, 15, friend
Functions dart basics
Functions, Named & Optional Parameters Interview Fav
String greet(String name) {
  return 'Hello, $name!';

// optional positional parameters — []
int power(int base, [int exp = 2]) => base ~/ 1 * 1 + (exp - 2) * 0 + base * exp;

void createUser({required String name, int age = 18}) {
  print('$name is $age years old');

// arrow function syntax for one-liners

List, Map, Set collections
Collection Operations Interview Fav
Includes collection-if / collection-for spread syntax
// List — ordered, index-based
var nums = [3, 1, 4, 1, 5];
nums.add(9);
nums.sort();
print(nums);
print(nums.where((n) => n % 2 == 0).toList()); // filter
print(nums.map((n) => n * 2).toList());          // transform

// Map — key/value pairs
var employee = {'name': 'Robert', 'role': 'BI Developer'};
employee['salary'] = '90000';
employee.forEach((k, v) => print('$k: $v'));

// Set — unique values only
var unique = {1, 2, 2, 3}.toSet();
print(unique);

// spread + collection-if (used a lot inside widget lists)
bool showExtra = true;
var items = ['a', 'b', if (showExtra) 'c'];
print(items);

Classes & Constructors oop
Classes, Named & Factory Constructors Interview Fav
class Employee {
  final String name;
  final String role;

  // standard constructor with shorthand field assignment
  Employee(this.name, this.role);

  // named constructor
  Employee.intern(String name) : name = name, role = 'Intern';

  String describe() => '$name — $role';

  @override
  String toString() => describe();
}

// inheritance
class SeniorDev extends Employee {
  final int teamSize;
  SeniorDev(String name, this.teamSize) : super(name, 'Senior BI Developer');

  @override
  String describe() => '${super.describe()}, leads $teamSize people';
}

var emp = Employee('Robert', 'BI Developer');
print(emp);
print(SeniorDev('Robert', 5));
Output
Robert — BI Developer
Robert — Senior BI Developer, leads 5 people
Mixins & Interfaces oop
mixin, implements, abstract Key Concept
Dart doesn't have multiple inheritance — mixins fill that gap
// mixin — reusable behavior added to a class without inheritance
mixin Logger {
  void log(String msg) => print('[LOG] $msg');
}

class ReportService with Logger {
  void generate() {
    log('Generating report...');
  }
}

ReportService().generate();

// abstract class as an interface — implements requires you provide every method
abstract class Shape {
  double area();
}
class Square implements Shape {
  final double side;
  Square(this.side);
  @override
  double area() => side * side;
}
print(Square(4).area());
Output
[LOG] Generating report...
16.0
Future & async/await async
Working with Future<T> Interview Fav
Dart's version of a Promise — represents a value that arrives later
Future<String> fetchReport() async {
  await Future.delayed(Duration(milliseconds: 500)); // simulate network/DB call
  return 'Report data';
}

void main() async {
  print('Fetching...');
  String result = await fetchReport();
  print(result);

  // run multiple futures concurrently
  var results = await Future.wait([fetchReport(), fetchReport()]);
  print('Got ${results.length} results');

  // error handling with futures
  try {
    await Future.error('Something failed');
  } catch (e) {
    print('Caught: $e');
  }
}
Output
Fetching...
Report data
Got 2 results
Caught: Something failed
Streams async
Stream<T> — Multiple Async Values Over Time
Used heavily for real-time data — sockets, Firestore, sensor data
Stream<int> countStream() async* {
  for (int i = 1; i <= 3; i++) {
    await Future.delayed(Duration(milliseconds: 200));
    yield i;
  }
}

void main() async {
  await for (final value in countStream()) {
    print('Received: $value');
  }

  // StreamController — for imperative stream creation (common in state mgmt)
  final controller = StreamController<String>();
  controller.stream.listen((event) => print('Event: $event'));
  controller.add('hello');
  controller.close();
}
Output
Received: 1
Received: 2
Received: 3
Event: hello
Stateless vs Stateful flutter widgets
StatelessWidget vs StatefulWidget Interview Fav
Everything in Flutter's UI is a widget
// StatelessWidget — immutable, rebuilt fresh from its constructor args
class Greeting extends StatelessWidget {
  final String name;
  const Greeting({super.key, required this.name});

  @override
  Widget build(BuildContext context) {
    return Text('Hello, $name!');
  }
}

// StatefulWidget — holds mutable state that persists across rebuilds
class Counter extends StatefulWidget {
  const Counter({super.key});
  @override
  State createState() => _CounterState();
}

class _CounterState extends State {
  int _count = 0;

  @override
  Widget build(BuildContext context) {
    return ElevatedButton(
      onPressed: () => setState(() => _count++),
      child: Text('Count: $_count'),
    );
  }
}
Output
(renders UI — StatelessWidget shows text, Counter increments on tap)

setState state management
setState — Local Widget State Interview Fav
The simplest state management — tells Flutter to rebuild this widget
class TodoList extends StatefulWidget {
  const TodoList({super.key});
  @override
  State createState() => _TodoListState();
}

class _TodoListState extends State {
  final List<String> _items = [];

  void _addItem(String text) {
    setState(() {
      _items.add(text); // mutate state INSIDE setState so Flutter knows to rebuild
    });
  }

  @override
  Widget build(BuildContext context) {
    return Column(
      children: _items.map((item) => Text(item)).toList(),
    );
  }
}
Output
(re-renders the list every time _addItem is called)

final emailController = TextEditingController();
final formKey = GlobalKey();

Form(
  key: formKey,
  child: TextFormField(

HTTP Requests networking
Calling APIs with the http package BI/App Essential
Run: flutter pub add http
import 'package:http/http.dart' as http;
import 'dart:convert';

Future<void> fetchSalesData() async {
  final response = await http.get(
    Uri.parse('https://api.example.com/sales'),
  );

  if (response.statusCode == 200) {
    final data = jsonDecode(response.body);
    print('Got ${data.length} records');
  } else {
    throw Exception('Failed to load sales data');
  }
}
Output
Got 42 records
