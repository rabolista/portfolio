Variables & Types basics
Declaring Variables Interview Fav
var, short declaration (:=), constants, zero values
package main

import "fmt"

func main() {
	// var with explicit type
	var name string = "Robert"
	var age int = 35

	// short declaration (type inferred) — most common
	role := "BI Developer"
	years := 15

	// constants
	const Pi = 3.14159

	// zero values — Go initializes unset vars automatically
	var count int     // 0
	var active bool  // false
	var label string // ""

	fmt.Println(name, age, role, years, Pi, count, active, label)
}
Output
Robert 35 BI Developer 15 3.14159 0 false
Strings & Formatting strings
fmt.Sprintf & strings package
Formatting verbs and common string operations
import (
	"fmt"
	"strings"
)

name := "Robert"
age := 35

// Sprintf verbs: %s string, %d int, %f float, %v any value, %T type
msg := fmt.Sprintf("%s is %d years old", name, age)
fmt.Println(msg)

// strings package
fmt.Println(strings.ToUpper(name))          // ROBERT
fmt.Println(strings.Contains(name, "ob")) // true
fmt.Println(strings.Split("a,b,c", ",")) // [a b c]
fmt.Println(strings.TrimSpace("  hi  ")) // "hi"
fmt.Println(strings.Join([]string{"a", "b"}, "-")) // a-b
Output
Robert is 35 years old
ROBERT
true
[a b c]
hi
a-b
If / Switch control flow
If Statements & Switch Interview Fav
No parentheses needed; switch doesn't fall through by default
score := 85

if score >= 90 {
	fmt.Println("A")
} else if score >= 80 {
	fmt.Println("B")
} else {
	fmt.Println("C")
}

// if with a short statement scoped to the if
if v := score * 2; v > 100 {
	fmt.Println("doubled over 100:", v)
}

// switch — no automatic fallthrough
switch role := "admin"; role {
case "admin":
	fmt.Println("Full access")
case "editor", "contributor":
	fmt.Println("Limited access")
default:
	fmt.Println("No access")
}
Output
B
doubled over 100: 170
Full access
For Loops control flow
Go's Only Loop — for Key Concept
Go has no while/do-while — for handles every loop pattern
// classic C-style for
for i := 0; i < 5; i++ {
	fmt.Print(i, " ")
}
fmt.Println()

// "while" style — just the condition
n := 0
for n < 3 {
	fmt.Println("n =", n)
	n++
}

// infinite loop with break
count := 0
for {
	if count == 2 {
		break
	}
	count++
}

// range — iterate over slices, maps, strings
fruits := []string{"apple", "banana"}
for i, fruit := range fruits {
	fmt.Println(i, fruit)
}
Output
0 1 2 3 4
n = 0
n = 1
n = 2
0 apple
1 banana
Arrays & Slices data structures
Slices — Go's Dynamic Arrays Interview Fav
Arrays are fixed-size; slices are what you'll actually use
// fixed-size array
var arr [3]int = [3]int{1, 2, 3}

// slice — dynamic, most common in Go code
nums := []int{3, 1, 4, 1, 5}
nums = append(nums, 9)      // add to end
fmt.Println(nums)              // [3 1 4 1 5 9]
fmt.Println(len(nums))         // 6
fmt.Println(nums[1:4])       // slice of a slice: [1 4 1]

// make() — pre-allocate a slice with length/capacity
scores := make([]int, 0, 10)
scores = append(scores, 85, 92)
fmt.Println(scores)

// 2D slice (like a table)
grid := [][]int{{1, 2}, {3, 4}}
fmt.Println(grid[1][0])  // 3
Output
[3 1 4 1 5 9]
6
[1 4 1]
[85 92]
3
Maps data structures
Map Operations Interview Fav
Go's key-value collection type
employee := map[string]string{
	"name": "Robert",
	"role": "BI Developer",
}

// access
fmt.Println(employee["name"])

// "comma ok" idiom — check if key exists
value, ok := employee["salary"]
fmt.Println(value, ok)  // "" false

// add / update
employee["salary"] = "90000"

// delete
delete(employee, "role")

// loop through a map
for k, v := range employee {
	fmt.Println(k, "=", v)
}

fmt.Println(len(employee))
Output
Robert
false
name = Robert
salary = 90000
2
Functions functions
Functions, Multiple Returns, Variadic Interview Fav
Go functions can return more than one value — very idiomatic
func greet(name string) string {
	return "Hello, " + name
}

// multiple return values — commonly (result, error)
func divide(a, b float64) (float64, error) {
	if b == 0 {
		return 0, fmt.Errorf("cannot divide by zero")
	}
	return a / b, nil
}

// variadic — accepts any number of args
func sum(nums ...int) int {
	total := 0
	for _, n := range nums {
		total += n
	}
	return total
}

fmt.Println(greet("Robert"))
result, err := divide(10, 2)
fmt.Println(result, err)
fmt.Println(sum(1, 2, 3, 4))
Output
Hello, Robert
5 <nil>
10
Closures functions
Function Literals & Closures
Functions can capture variables from their surrounding scope
func counter() func() int {
	count := 0
	return func() int {
		count++
		return count
	}
}

next := counter()
fmt.Println(next())  // 1
fmt.Println(next())  // 2
fmt.Println(next())  // 3

// anonymous function, called immediately
func() {
	fmt.Println("ran immediately")
}()
Output
1
2
3
ran immediately
Structs structs & interfaces
Struct Basics Interview Fav
Go's way of grouping fields — no classes, no inheritance
type Employee struct {
	Name   string
	Role   string
	Salary float64
}

// create with field names (recommended)
emp := Employee{Name: "Robert", Role: "BI Developer", Salary: 90000}

// access / update fields
fmt.Println(emp.Name)
emp.Salary *= 1.1
fmt.Println(emp.Salary)

// nested structs
type Address struct {
	City    string
	Country string
}
type Person struct {
	Employee
	Home Address
}
p := Person{Employee: emp, Home: Address{City: "London", Country: "Canada"}}
fmt.Println(p.Name, p.Home.City) // embedded field promotion
Output
Robert
99000
Robert London
Methods & Receivers structs & interfaces
Value vs Pointer Receivers Key Concept
Pointer receivers can modify the struct; value receivers work on a copy
type Account struct {
	Balance float64
}

// value receiver — read-only, gets a copy
func (a Account) Describe() string {
	return fmt.Sprintf("Balance: $%.2f", a.Balance)
}

// pointer receiver — can mutate the original struct
func (a *Account) Deposit(amount float64) {
	a.Balance += amount
}

acc := Account{Balance: 100}
acc.Deposit(50)  // Go auto-takes the address here
fmt.Println(acc.Describe())
Output
Balance: $150.00
Interfaces structs & interfaces
Interfaces — Implicit Implementation Interview Fav
No "implements" keyword — any type with the right methods satisfies an interface
type Shape interface {
	Area() float64
}

type Rectangle struct { Width, Height float64 }
func (r Rectangle) Area() float64 { return r.Width * r.Height }

type Circle struct { Radius float64 }
func (c Circle) Area() float64 { return 3.14159 * c.Radius * c.Radius }

func printArea(s Shape) {
	fmt.Printf("Area: %.2f\n", s.Area())
}

printArea(Rectangle{Width: 4, Height: 5})
printArea(Circle{Radius: 3})

// empty interface — accepts any type (like interface{} / any)
var anything any = "could be anything"
fmt.Println(anything)
Output
Area: 20.00
Area: 28.27
could be anything
Error Handling error handling
The error Type — No Exceptions Interview Fav
Go handles errors as regular return values, not try/catch
import (
	"errors"
	"fmt"
)

func validateAge(age int) error {
	if age < 0 {
		return errors.New("age cannot be negative")
	}
	return nil
}

if err := validateAge(-5); err != nil {
	fmt.Println("Error:", err)
}

// custom error type
type ValidationError struct {
	Field string
}
func (e *ValidationError) Error() string {
	return fmt.Sprintf("invalid field: %s", e.Field)
}

func checkField(name string) error {
	if name == "" {
		return &ValidationError{Field: "name"}
	}
	return nil
}

err := checkField("")
fmt.Println(err)
Output
Error: age cannot be negative
invalid field: name
Defer / Panic / Recover error handling
defer, panic, recover Key Concept
defer always runs; panic/recover is Go's rare "exception-like" mechanism
func process() {
	defer fmt.Println("cleanup runs last")  // runs when function returns
	fmt.Println("processing...")
}

// recover from a panic so the program doesn't crash
func safeCall() {
	defer func() {
		if r := recover(); r != nil {
			fmt.Println("recovered from:", r)
		}
	}()
	panic("something went wrong")
}

process()
safeCall()
fmt.Println("program continues")
Output
processing...
cleanup runs last
recovered from: something went wrong
program continues
Goroutines concurrency
Starting Goroutines Interview Fav
Lightweight threads managed by the Go runtime
import (
	"fmt"
	"sync"
)

func worker(id int, wg *sync.WaitGroup) {
	defer wg.Done()
	fmt.Printf("worker %d done\n", id)
}

func main() {
	var wg sync.WaitGroup

	for i := 1; i <= 3; i++ {
		wg.Add(1)
		go worker(i, &wg)  // "go" starts it concurrently
	}

	wg.Wait() // block until all goroutines call Done()
	fmt.Println("all workers finished")
}
Output
worker 3 done
worker 1 done
worker 2 done
all workers finished
(order may vary — that's concurrency!)
Channels concurrency
Sending & Receiving on Channels Interview Fav
How goroutines communicate safely
ch := make(chan string)

go func() {
	ch <- "hello from goroutine"  // send
}()

msg := <-ch  // receive (blocks until a value arrives)
fmt.Println(msg)

// buffered channel — doesn't block until full
results := make(chan int, 3)
results <- 1
results <- 2
close(results)  // signal no more values

for r := range results {
	fmt.Println("result:", r)
}
Output
hello from goroutine
result: 1
result: 2
Select concurrency
select — Wait on Multiple Channels
Like a switch statement, but for channels
import (
	"fmt"
	"time"
)

c1 := make(chan string)
c2 := make(chan string)

go func() { time.Sleep(50 * time.Millisecond); c1 <- "one" }()
go func() { time.Sleep(100 * time.Millisecond); c2 <- "two" }()

for i := 0; i < 2; i++ {
	select {
	case msg1 := <-c1:
		fmt.Println("received", msg1)
	case msg2 := <-c2:
		fmt.Println("received", msg2)
	}
}
Output
received one
received two
Pointers advanced
Pointers — & and * Interview Fav
Pass by reference instead of copying values
x := 10
p := &x         // p is a pointer to x's address

fmt.Println(*p)     // 10 — dereference to get the value
*p = 20       // modify x through the pointer
fmt.Println(x)      // 20

// why it matters: functions that need to modify the caller's value
func double(n *int) {
	*n *= 2
}
double(&x)
fmt.Println(x)  // 40
Output
10
20
40
Generics advanced
Generic Functions (Go 1.18+)
Write one function that works across multiple types
func Max[T int | float64](a, b T) T {
	if a > b {
		return a
	}
	return b
}

fmt.Println(Max(3, 7))         // 7 (ints)
fmt.Println(Max(2.5, 1.1))     // 2.5 (floats)

// generic slice helper
func Filter[T any](items []T, keep func(T) bool) []T {
	var result []T
	for _, item := range items {
		if keep(item) {
			result = append(result, item)
		}
	}
	return result
}

nums := []int{1, 2, 3, 4, 5, 6}
evens := Filter(nums, func(n int) bool { return n%2==0 })
fmt.Println(evens)
Output
7
2.5
[2 4 6]
JSON advanced
encoding/json — Marshal & Unmarshal BI Essential
Struct tags control the JSON field names
import (
	"encoding/json"
	"fmt"
)

type Employee struct {
	Name string  `json:"name"`
	Role string  `json:"role"`
}

emp := Employee{Name: "Robert", Role: "BI Developer"}

// struct -> JSON
data, _ := json.Marshal(emp)
fmt.Println(string(data))

// JSON -> struct
raw := []byte(`{"name":"Alice","role":"Analyst"}`)
var parsed Employee
json.Unmarshal(raw, &parsed)
fmt.Println(parsed.Name, parsed.Role)
Output
{"name":"Robert","role":"BI Developer"}
Alice Analyst
