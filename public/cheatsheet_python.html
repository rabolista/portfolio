# A normal function returns once. A generator yields many times.
def batch_ids(start, end, step):
    n = start
    while n < end:
        yield n
        n += step

for batch_id in batch_ids(0, 10000, 2500):
    print({batch_id")

Decorators functions
@decorator Syntax Interview Fav
A function that wraps another function to add behaviour
import time

# A decorator that times any function it's applied to
def timer(func):
    def wrapper(*args, **kwargs):
        start  = time.time()
        result  = func(*args, **kwargs)
        print({func.__name__ took {time.time() - start:.4f}s")
        return result
    return wrapper

@timer
def load_report(rows):
    return [r for r in range(rows)]

data  = load_report(1000000)
print(len(data))
Output
load_report took 0.0421s
