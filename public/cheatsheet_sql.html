SELECT & WHERE basics
Basic Queries & Filtering Interview Fav
SELECT ProductID, ProductName, Sales
FROM dbo.SalesFact
WHERE Sales > 1000
  AND Region = 'East';

-- multiple conditions
SELECT * FROM dbo.Employees
WHERE Department IN ('IT', 'HR')
  AND Salary BETWEEN 60000 AND 100000;

-- pattern matching
SELECT * FROM dbo.Customers
WHERE Email LIKE '%@masco.com';

-- column alias
SELECT ProductName AS Product, Sales AS Revenue
FROM dbo.SalesFact;
Output
(returns matching rows from SalesFact / Employees / Customers)
ORDER BY & TOP basics
Sorting & Limiting Rows Interview Fav
-- sort descending, then ascending as a tiebreaker
SELECT * FROM dbo.SalesFact
ORDER BY Sales DESC, ProductName ASC;

-- TOP N (SQL Server syntax — MySQL/Postgres use LIMIT)
SELECT TOP 10 * FROM dbo.SalesFact
ORDER BY Sales DESC;

-- TOP with ties — includes rows tied with the last one
SELECT TOP 5 WITH TIES * FROM dbo.SalesFact
ORDER BY Sales DESC;

-- paging with OFFSET / FETCH (SQL Server 2012+)
SELECT * FROM dbo.SalesFact
ORDER BY SaleDate
OFFSET 20 ROWS FETCH NEXT 10 ROWS ONLY;
Output
(returns sorted / paged rows)
DISTINCT & NULL basics
DISTINCT, ISNULL, COALESCE Interview Fav
-- unique values only
SELECT DISTINCT Department FROM dbo.Employees;

-- NULL checks — never use = NULL, always IS NULL
SELECT * FROM dbo.Orders WHERE ShippedDate IS NULL;
SELECT * FROM dbo.Orders WHERE ShippedDate IS NOT NULL;

-- ISNULL — replace a null with a default (SQL Server specific)
SELECT ProductName, ISNULL(Discount, 0) AS Discount
FROM dbo.Products;

-- COALESCE — first non-null value from a list (ANSI standard, works everywhere)
SELECT COALESCE(NickName, FirstName, 'Unknown') AS DisplayName
FROM dbo.Customers;
Output
(returns rows with null-safe defaults applied)
INNER / LEFT / RIGHT / FULL joins
All Join Types Interview Fav
The single most-asked SQL interview topic
-- INNER JOIN — only matching rows in both tables
SELECT o.OrderID, c.CustomerName
FROM dbo.Orders o
INNER JOIN dbo.Customers c ON o.CustomerID = c.CustomerID;

-- LEFT JOIN — all rows from the left table, matched or not
SELECT c.CustomerName, o.OrderID
FROM dbo.Customers c
LEFT JOIN dbo.Orders o ON c.CustomerID = o.CustomerID;

-- find customers with NO orders (classic anti-join pattern)
SELECT c.CustomerName
FROM dbo.Customers c
LEFT JOIN dbo.Orders o ON c.CustomerID = o.CustomerID
WHERE o.OrderID IS NULL;

-- RIGHT JOIN — all rows from the right table
SELECT * FROM dbo.Orders o
RIGHT JOIN dbo.Customers c ON o.CustomerID = c.CustomerID;

-- FULL OUTER JOIN — all rows from both sides
SELECT * FROM dbo.Orders o
FULL OUTER JOIN dbo.Customers c ON o.CustomerID = c.CustomerID;
Output
(returns joined result sets per the join type)
UNION vs UNION ALL joins
Combining Result Sets Common Gotcha
UNION removes duplicates (slower); UNION ALL keeps everything (faster)
-- both queries must return the same number/type of columns
SELECT CustomerName, 'Current' AS Status FROM dbo.ActiveCustomers
UNION
SELECT CustomerName, 'Archived' AS Status FROM dbo.ArchivedCustomers;

-- UNION ALL — keeps duplicates, much cheaper performance-wise
SELECT ProductID FROM dbo.SalesFact_2025
UNION ALL
SELECT ProductID FROM dbo.SalesFact_2026;
Output
(combined result set, deduplicated for UNION, not for UNION ALL)
GROUP BY & HAVING aggregation
Grouping & Filtering Groups Interview Fav
WHERE filters rows before grouping; HAVING filters groups after
-- total sales per product
SELECT ProductName, SUM(Sales) AS TotalSales
FROM dbo.SalesFact
GROUP BY ProductName;

-- HAVING — filter AFTER aggregation (WHERE can't reference aggregates)
SELECT ProductName, SUM(Sales) AS TotalSales
FROM dbo.SalesFact
GROUP BY ProductName
HAVING SUM(Sales) > 50000;

-- group by multiple columns
SELECT Region, ProductCategory, COUNT(*) AS OrderCount
FROM dbo.SalesFact
GROUP BY Region, ProductCategory
ORDER BY OrderCount DESC;
Output
(returns grouped/aggregated rows)
Aggregate Functions aggregation
COUNT, SUM, AVG, MIN, MAX
SELECT
    COUNT(*)          AS TotalOrders,
    COUNT(DISTINCT CustomerID) AS UniqueCustomers,
    SUM(Sales)         AS TotalRevenue,
    AVG(Sales)         AS AvgOrderValue,
    MIN(Sales)         AS SmallestOrder,
    MAX(Sales)         AS LargestOrder
FROM dbo.SalesFact;
Output
TotalOrders: 15420 | UniqueCustomers: 3200 | TotalRevenue: 2,450,000.00 | AvgOrderValue: 158.90 | SmallestOrder: 4.99 | LargestOrder: 12,400.00
Subqueries subqueries & CTEs
Subqueries in WHERE, FROM, SELECT Interview Fav
-- subquery in WHERE
SELECT ProductName FROM dbo.Products
WHERE ProductID IN (
    SELECT ProductID FROM dbo.SalesFact WHERE Sales > 10000
);

-- correlated subquery — references the outer query
SELECT e.Name, e.Salary
FROM dbo.Employees e
WHERE e.Salary > (
    SELECT AVG(e2.Salary) FROM dbo.Employees e2 WHERE e2.Department = e.Department
);

-- subquery in FROM (derived table)
SELECT Department, AvgSalary
FROM (
    SELECT Department, AVG(Salary) AS AvgSalary
    FROM dbo.Employees
    GROUP BY Department
) AS DeptAverages
WHERE AvgSalary > 70000;

-- EXISTS — often faster than IN for existence checks
SELECT c.CustomerName FROM dbo.Customers c
WHERE EXISTS (SELECT 1 FROM dbo.Orders o WHERE o.CustomerID = c.CustomerID);
Output
(returns filtered/derived rows)
CTEs — WITH subqueries & CTEs
Common Table Expressions BI Essential
Cleaner than nested subqueries, and required for recursive queries
-- basic CTE — makes multi-step logic readable
WITH DeptAverages AS (
    SELECT Department, AVG(Salary) AS AvgSalary
    FROM dbo.Employees
    GROUP BY Department
)
SELECT e.Name, e.Department, d.AvgSalary
FROM dbo.Employees e
JOIN DeptAverages d ON e.Department = d.Department
WHERE e.Salary > d.AvgSalary;

-- multiple CTEs chained together
WITH HighSales AS (
    SELECT ProductID, SUM(Sales) AS Total FROM dbo.SalesFact GROUP BY ProductID
),
TopProducts AS (
    SELECT TOP 5 * FROM HighSales ORDER BY Total DESC
)
SELECT p.ProductName, t.Total
FROM TopProducts t
JOIN dbo.Products p ON t.ProductID = p.ProductID;

-- recursive CTE — walking a hierarchy (e.g., org chart)
WITH OrgChart AS (
    SELECT EmployeeID, ManagerID, Name, 0 AS Level
    FROM dbo.Employees WHERE ManagerID IS NULL
    UNION ALL
    SELECT e.EmployeeID, e.ManagerID, e.Name, o.Level + 1
    FROM dbo.Employees e
    JOIN OrgChart o ON e.ManagerID = o.EmployeeID
)
SELECT * FROM OrgChart ORDER BY Level;
Output
(returns readable multi-step / hierarchical results)
ROW_NUMBER, RANK, OVER window functions
Window Functions Interview Fav
Calculate across a set of rows without collapsing them like GROUP BY does
-- ROW_NUMBER — unique sequential number per partition
SELECT ProductName, Region, Sales,
    ROW_NUMBER() OVER (PARTITION BY Region ORDER BY Sales DESC) AS RowNum
FROM dbo.SalesFact;

-- RANK / DENSE_RANK — ties share the same rank (RANK skips numbers after a tie)
SELECT ProductName, Sales,
    RANK()       OVER (ORDER BY Sales DESC) AS SalesRank,
    DENSE_RANK() OVER (ORDER BY Sales DESC) AS DenseSalesRank
FROM dbo.SalesFact;

-- running total — classic BI pattern
SELECT SaleDate, Sales,
    SUM(Sales) OVER (ORDER BY SaleDate ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS RunningTotal
FROM dbo.SalesFact;

-- LAG / LEAD — compare to previous/next row (e.g., month-over-month)
SELECT SaleMonth, Revenue,
    LAG(Revenue) OVER (ORDER BY SaleMonth) AS PrevMonthRevenue,
    Revenue - LAG(Revenue) OVER (ORDER BY SaleMonth) AS Change
FROM dbo.MonthlyRevenue;

-- use ROW_NUMBER to pick the top row per group ("greatest-n-per-group")
WITH Ranked AS (
    SELECT *, ROW_NUMBER() OVER (PARTITION BY Region ORDER BY Sales DESC) AS rn
    FROM dbo.SalesFact
)
SELECT * FROM Ranked WHERE rn = 1;
Output
(returns ranked/running-total rows without collapsing detail)
INSERT / UPDATE / DELETE insert / update / delete
Data Manipulation Statements Interview Fav
-- INSERT single row
INSERT INTO dbo.Employees (Name, Department, Salary)
VALUES ('Robert', 'IT', 90000);

-- INSERT multiple rows
INSERT INTO dbo.Employees (Name, Department, Salary)
VALUES
    ('Alice', 'HR', 70000),
    ('Bob', 'IT', 85000);

-- INSERT from a SELECT (great for staging -> warehouse loads)
INSERT INTO dbo.EmployeeArchive (Name, Department, Salary)
SELECT Name, Department, Salary FROM dbo.Employees WHERE Terminated = 1;

-- UPDATE — always use WHERE unless you mean to update every row!
UPDATE dbo.Employees
SET Salary = Salary * 1.1
WHERE Department = 'IT';

-- DELETE — same warning applies
DELETE FROM dbo.Employees
WHERE Terminated = 1 AND TerminationDate < DATEADD(YEAR, -2, GETDATE());

-- TRUNCATE — faster than DELETE, wipes the whole table, resets identity
TRUNCATE TABLE dbo.StagingImport;
Output
(rows affected — always check row counts before running UPDATE/DELETE in prod)
MERGE (Upsert) insert / update / delete
MERGE — Insert or Update in One Statement ETL Essential
Very common in SSIS/ETL loads to sync a staging table into a dimension table
MERGE dbo.DimCustomer AS target
USING dbo.StagingCustomer AS source
ON target.CustomerID = source.CustomerID

WHEN MATCHED THEN
    UPDATE SET
        target.CustomerName = source.CustomerName,
        target.Email = source.Email

WHEN NOT MATCHED BY TARGET THEN
    INSERT (CustomerID, CustomerName, Email)
    VALUES (source.CustomerID, source.CustomerName, source.Email)

WHEN NOT MATCHED BY SOURCE THEN
    DELETE;
Output
(syncs DimCustomer to match StagingCustomer exactly)
CASE Expressions functions
CASE WHEN — SQL's if/else Interview Fav
SELECT ProductName, Sales,
    CASE
        WHEN Sales >= 10000 THEN 'High'
        WHEN Sales >= 5000  THEN 'Medium'
        ELSE 'Low'
    END AS SalesTier
FROM dbo.SalesFact;

-- conditional aggregation — pivot-like counts in one row
SELECT
    SUM(CASE WHEN Region = 'East' THEN Sales ELSE 0 END) AS EastSales,
    SUM(CASE WHEN Region = 'West' THEN Sales ELSE 0 END) AS WestSales
FROM dbo.SalesFact;
Output
EastSales: 1,200,000.00 | WestSales: 980,000.00
String Functions functions
CONCAT, SUBSTRING, TRIM, LEN
SELECT
    CONCAT(FirstName, ' ', LastName)      AS FullName,
    UPPER(Email)                          AS EmailUpper,
    LEN(ProductName)                      AS NameLength,
    SUBSTRING(ProductCode, 1, 3)             AS CodePrefix,
    TRIM('  padded  ')                    AS Trimmed,
    REPLACE(Phone, '-', '')                AS PhoneDigitsOnly,
    LEFT(ProductName, 5)                  AS First5Chars,
    RIGHT(ProductCode, 4)                 AS Last4Chars
FROM dbo.Customers c
JOIN dbo.Products p ON 1=1;
Output
(returns formatted/cleaned text columns)
Date Functions functions
GETDATE, DATEADD, DATEDIFF, FORMAT BI Essential
SELECT
    GETDATE()                                  AS Now,
    DATEADD(DAY, 30, OrderDate)                AS DueDate,
    DATEDIFF(DAY, OrderDate, ShippedDate)    AS DaysToShip,
    FORMAT(OrderDate, 'yyyy-MM-dd')             AS FormattedDate,
    YEAR(OrderDate)                            AS OrderYear,
    MONTH(OrderDate)                           AS OrderMonth,
    DATENAME(WEEKDAY, OrderDate)              AS DayName
FROM dbo.Orders;

-- date truncation — group by month
SELECT
    DATEFROMPARTS(YEAR(OrderDate), MONTH(OrderDate), 1) AS MonthStart,
    SUM(Sales) AS MonthlySales
FROM dbo.SalesFact
GROUP BY DATEFROMPARTS(YEAR(OrderDate), MONTH(OrderDate), 1);
Output
(returns calculated/grouped date columns)
Views views & procedures
CREATE VIEW BI Essential
A saved, reusable query — great for exposing clean semantic layers to Power BI
CREATE VIEW dbo.vw_SalesSummary AS
SELECT
    p.ProductName,
    r.RegionName,
    SUM(s.Sales) AS TotalSales
FROM dbo.SalesFact s
JOIN dbo.Products p ON s.ProductID = p.ProductID
JOIN dbo.Regions r ON s.RegionID = r.RegionID
GROUP BY p.ProductName, r.RegionName;

-- query it just like a table
SELECT * FROM dbo.vw_SalesSummary WHERE TotalSales > 100000;

-- update / drop
ALTER VIEW dbo.vw_SalesSummary AS SELECT * FROM dbo.SalesFact;
DROP VIEW dbo.vw_SalesSummary;
Output
(creates/queries a reusable virtual table)
Stored Procedures views & procedures
CREATE PROCEDURE — Parameterized, Reusable Logic BI Essential
CREATE PROCEDURE dbo.usp_GetSalesByRegion
    @RegionName NVARCHAR(50),
    @StartDate  DATE,
    @EndDate    DATE
AS
BEGIN
    SET NOCOUNT ON;

    SELECT ProductName, SUM(Sales) AS TotalSales
    FROM dbo.SalesFact s
    JOIN dbo.Regions r ON s.RegionID = r.RegionID
    WHERE r.RegionName = @RegionName
      AND s.SaleDate BETWEEN @StartDate AND @EndDate
    GROUP BY ProductName;
END;

-- execute it
EXEC dbo.usp_GetSalesByRegion
    @RegionName = 'East',
    @StartDate = '2026-01-01',
    @EndDate = '2026-06-30';
Output
(returns filtered sales for the given region/date range)
Indexes indexes & transactions
CREATE INDEX — Speeding Up Queries Performance
Clustered = physical row order (one per table); Nonclustered = a separate lookup structure
-- nonclustered index on a frequently filtered column
CREATE NONCLUSTERED INDEX IX_SalesFact_ProductID
ON dbo.SalesFact (ProductID);

-- covering index — includes extra columns so SQL doesn't need to look up the row
CREATE NONCLUSTERED INDEX IX_SalesFact_Region_Covering
ON dbo.SalesFact (RegionID)
INCLUDE (ProductID, Sales);

-- check how a query uses (or ignores) indexes
SET STATISTICS IO ON;
SELECT * FROM dbo.SalesFact WHERE ProductID = 1001;

DROP INDEX IX_SalesFact_ProductID ON dbo.SalesFact;
Output
(subsequent queries on ProductID become an index seek instead of a table scan)
Transactions indexes & transactions
BEGIN TRAN / COMMIT / ROLLBACK Interview Fav
Keeps multi-step writes all-or-nothing
BEGIN TRANSACTION;

BEGIN TRY
    UPDATE dbo.Accounts SET Balance = Balance - 500 WHERE AccountID = 1;
    UPDATE dbo.Accounts SET Balance = Balance + 500 WHERE AccountID = 2;

    COMMIT TRANSACTION;
    PRINT 'Transfer succeeded';
END TRY
BEGIN CATCH
    ROLLBACK TRANSACTION;
    PRINT ERROR_MESSAGE();
END CATCH;
Output
Transfer succeeded — or fully rolled back if either UPDATE fails
